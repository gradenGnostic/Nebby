cmake_minimum_required(VERSION 3.20)
project(MoonNativeFilesystem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(ctr_module "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/external/triaevum/runtime/triaevum_module")
add_library(ctr_native_filesystem SHARED bridge/ctr_native_filesystem.cpp
  "${ctr_module}/src/rooted_filesystem_backend.cpp")
target_include_directories(ctr_native_filesystem PRIVATE "${ctr_module}/include")
target_compile_options(ctr_native_filesystem PRIVATE -Wall -Wextra -Werror)
add_executable(ctr_native_filesystem_tests tests/rooted_persistence.cpp)
target_include_directories(ctr_native_filesystem_tests PRIVATE "${ctr_module}/include")
target_link_libraries(ctr_native_filesystem_tests PRIVATE ctr_native_filesystem)
target_compile_options(ctr_native_filesystem_tests PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME ctr_native_filesystem_persistence COMMAND ctr_native_filesystem_tests)
